## Regenerating the outputs

```
cd "KiCad project"
kicad-cli sch export pdf -o ../fabrication/RevA-<date>/ScaleController_RevA-schematic.pdf ScaleController_RevA.kicad_sch
kicad-cli sch export bom -o ../fabrication/RevA-<date>/ScaleController_RevA-BOM.csv \
  --fields 'Reference,${QUANTITY},Value,Footprint,MF,Manufacturer_Name,MANUFACTURER,MP,Manufacturer_Part_Number,Mouser Part Number,Description,Datasheet,${DNP}' \
  --group-by 'Value,Footprint,MP,Manufacturer_Part_Number' --ref-range-delimiter '' ScaleController_RevA.kicad_sch
kicad-cli pcb export gerbers -o ../fabrication/RevA-<date>/gerbers/ \
  -l F.Cu,In1.Cu,In2.Cu,B.Cu,F.Paste,B.Paste,F.SilkS,B.SilkS,F.Mask,B.Mask,Edge.Cuts --subtract-soldermask ScaleController_RevA.kicad_pcb
kicad-cli pcb export drill -o ../fabrication/RevA-<date>/gerbers/ --excellon-separate-th --excellon-units mm --generate-map ScaleController_RevA.kicad_pcb
kicad-cli pcb export pos -o ../fabrication/RevA-<date>/ScaleController_RevA-pos-top.csv --side front --format csv --units mm --exclude-dnp ScaleController_RevA.kicad_pcb
kicad-cli pcb export pos -o ../fabrication/RevA-<date>/ScaleController_RevA-pos-bottom.csv --side back --format csv --units mm --exclude-dnp ScaleController_RevA.kicad_pcb
```
